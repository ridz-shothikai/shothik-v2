  </Box>

Option B: Click-to-Play Modal (BEST for Control)
const [selectedVideo, setSelectedVideo] = useState(null);
<Card onClick={() => setSelectedVideo(metric.videoUrl)}>
  <Box sx={{ position: 'relative' }}>
    <img src={metric.thumbnail} alt="" />
    <PlayCircleIcon sx={{ 
      position: 'absolute',
      top: '50%',
      left: '50%',
      transform: 'translate(-50%, -50%)',
      fontSize: 64,
      color: 'primary.main'
    }} />
  </Box>
  {/* Metric content */}
</Card>
{/* Modal with full video */}
<Dialog open={!!selectedVideo} onClose={() => setSelectedVideo(null)}>
  <video src={selectedVideo} controls autoPlay />
</Dialog>
Result: Click card → Full-screen video plays with controls
